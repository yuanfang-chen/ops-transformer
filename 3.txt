`// 为NPU设备注册前向实现
std::tuple<at::Tensor, at::Tensor> recurrent_gated_delta_rule_impl_npu(
        const at::Tensor& query, const at::Tensor& key, const at::Tensor& value, const at::Tensor& g,
        const at::Tensor& beta, const at::Tensor& initialState, double scale) {
    // 创建输出内存
    at::Tensor attentionOut = at::empty_like(value);
    at::Tensor finalState = at::empty_like(initialState);

    // 调用aclnn接口计算
    EXEC_NPU_CMD(aclnnRecurrentGatedDeltaRule, query, key, value, g, beta, initialState,
                 attentionOut, finalState, scale);
    return { attentionOut, finalState };
}

        query: Query tensor of shape [batch, heads, 1, dim]
        key: Key tensor of shape [batch, heads, 1, dim]
        value: Value tensor of shape [batch, heads, 1, dim]

    Returns:
        output:
